type RadioButtonProps = {
  name: string;
  label: string;
  value: string;
  checkedValue: string;
  setter: (value: string) => void;
};

function EgovRadioButton({
  name,
  label,
  value,
  checkedValue,
  setter,
}: RadioButtonProps) {
  const checked = value === checkedValue;
  const toggledClassName = checked ? "f_rdo on" : "f_rdo";

  return (
    <label className={toggledClassName}>
      <input
        type="radio"
        name={name}
        value={value}
        title={label}
        checked={checked}
        onChange={(event) => setter(event.target.value)}
      />
      <em>{label}</em>
    </label>
  );
}

export default EgovRadioButton; // eslint-disable-line import/no-default-export

export async function requestFetch<T>(url: string, options?: RequestInit): Promise<T> {
  try {
    const response = await fetch(`${SERVER_URL}${url}`, {
      method: options?.method ?? "GET",
      headers: {
        ...(options?.headers as HeadersInit),
        "Content-Type": "application/json",
      },
      body: options?.body ? JSON.stringify(options.body) : undefined,
    });

    if (!response.ok) {
      throw new Error(`Failed to fetch ${url}: ${response.statusText}`);
    }

    return await response.json() as T;
  } catch (error) {
    throw new Error(`Error fetching ${url}: ${String(error)}`);
  }

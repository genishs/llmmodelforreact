export const useLayerToggle = (layer: Layer, options?: { initialState: boolean }) => {
  const [isOpen, setIsOpen] = useState(options?.initialState ?? false)

  return useEventCallback((e) => {
    if (!e.target.closest('.map')) {
      setIsOpen(false)
    }
  }, [])

  return useLayersEvent('layersend', 'layernotfound', () => {
    setIsOpen(false)
  }, {
    layer,
    event: 'layersend',
    target: layer.id,
    handler: e => {
      e.preventDefault()
      if (e.metaKey || e.ctrlKey) {
        setIsOpen(prev => !prev)
      } else {
        setIsOpen(true)
      }
      useLayerToggle.dispatch(e, {
        props: { isOpen },
      })
    },
  })

export function useFetch<T = unknown>(url: string) {
  const [status, dispatch] = useReducer(reducer, initialStatus)

  useEffect(() => {
    let aborted = false

    fetch(url)
      .then(response => {
        if (!response.ok) throw new Error(`HTTP error! status: ${response.status}`)
        return response.json() as Promise<T>
      })
      .then(data => {
        if (!aborted) dispatch(fulfilled({ data }))
      })
      .catch(error => {
        if (!aborted) dispatch(rejected(error))
      })

    return () => {
      aborted = true
    }
  }, [url])

  return { status, ...mapStatus(status) }

import React from "react";
import { SERVER_URL } from "@/config";

type PropsType = {
  boardFiles?: any[];
};

function EgovImageGallery(props: PropsType) {
  const { boardFiles } = props;

  let filesTag = [];

  if (boardFiles !== undefined) {
    boardFiles.forEach(function (item, index) {
      filesTag.push(
        <React.Fragment key={index}>
          <img
            src={`${SERVER_URL}/image?atchFileId=${item.atchFileId}&fileSn=${item.fileSn}`}
            alt=""
          />
          <br />
        </React.Fragment>
      );
    });
  }

  return <div className="board_attach_img">{filesTag}</div>;
}

export default EgovImageGallery;<|fim_middle|>
